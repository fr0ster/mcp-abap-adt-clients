form get_field
     using langu type sy-langu
           tabname type dfies-tabname
           fieldname type dfies-fieldname
     changing shstruc type shstruc.
  statics dfies_tab type table of dfies.
  data dfies type dfies.

  read table dfies_Tab into dfies with key
       tabname = tabname
       fieldname = fieldname
       binary search.
  if sy-subrc <> 0.
    clear dfies_tab[].
    CALL FUNCTION 'DDIF_FIELDINFO_GET'
         EXPORTING
              tabname        = tabname
              LANGU          = LANGU
         TABLES
              DFIES_TAB      = dfies_tab
         EXCEPTIONS
              OTHERS         = 3.
    sort dfies_tab by tabname fieldname.
    read table dfies_Tab into dfies with key
         tabname = tabname
         fieldname = fieldname
         binary search.
  endif.
  check sy-subrc = 0.

  SHSTRUC-TABNAME = dfies-tabname.
  SHSTRUC-FIELDNAME = dfies-fieldname.
  SHSTRUC-FIELDTYPE = dfies-DATATYPE.
  SHSTRUC-FIELDTEXT = dfies-FIELDTEXT.
  IF dfies-MASKLEN <> 0.
    SHSTRUC-FIELDLEN = dfies-MASKLEN.
  ELSE.
    SHSTRUC-FIELDLEN = dfies-OUTPUTLEN.
  ENDIF.
  SHSTRUC-KEYWORD = dfies-REPTEXT.
  SHSTRUC-KEYWORDLEN = STRLEN( SHSTRUC-KEYWORD ).
  SHSTRUC-SCRTEXT_S = dfies-SCRTEXT_S.
  SHSTRUC-SCRLEN_S = STRLEN( SHSTRUC-SCRTEXT_S ).
  SHSTRUC-SCRTEXT_M = dfies-SCRTEXT_M.
  SHSTRUC-SCRLEN_M = STRLEN( SHSTRUC-SCRTEXT_M ).
  SHSTRUC-SCRTEXT_L = dfies-SCRTEXT_L.
  SHSTRUC-SCRLEN_L = STRLEN( SHSTRUC-SCRTEXT_L ).
  SHSTRUC-KEYFLAG = SPACE.

endform.
